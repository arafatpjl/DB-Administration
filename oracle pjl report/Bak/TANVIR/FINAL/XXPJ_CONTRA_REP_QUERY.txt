SELECT DISTINCT
       fnd_flex_ext.get_segs ('SQLGL',
                              'GL#',
                              xgl.chart_of_accounts_id,
                              xal.code_combination_id)
          ACCOUNT,
       xxpj_get_cc_desc(xal.code_combination_id) account_description,
       xal.accounted_dr,
       xal.accounted_cr,
       xal.entered_dr,
       xal.entered_cr,
       srcba.bank_account_num source_bank_account_num,
       srcba.bank_account_name source_bank_account_name,
       srcba.currency_code source_currency_code,
       srcbb.bank_name source_bank_name,
       srcbb.bank_branch_name source_bank_branch_name,
       srcle.name SOURCE_LEGAL_ENTITY_NAME,
       destba.bank_account_num destination_bank_account_num,
       destba.bank_account_name destination_bank_account_name,
       destba.currency_code destination_currency_code,
       destbb.bank_name destination_bank_name,
       destbb.bank_branch_name destination_bank_branch_name,
       destle.name DESTINATION_LEGAL_ENTITY_NAME,
       cpt.trxn_reference_number ,
       cpt.TRANSACTION_DATE
  FROM ce_cashflows ca,
       ce_payment_transactions cpt,
       ce_bank_accounts srcba,
       ce_bank_accounts destba,
       ce_bank_branches_v srcbb,
       ce_bank_branches_v destbb,
       xle_entity_profiles srcle,
       xle_entity_profiles destle,
       xla_events xe,
       xla.xla_transaction_entities xte,
       xla_ae_headers xah,
       XLA_AE_LINES xal,
       xla_gl_ledgers_v xgl,
       gl_daily_conversion_types gdct,
       xla_lookups xlk,
       gl_encumbrance_types get
 WHERE     ca.trxn_reference_number = cpt.trxn_reference_number
       AND cpt.trxn_status_code = 'SETTLED'
       AND srcba.bank_account_id = cpt.source_bank_account_id
       AND srcbb.branch_party_id = srcba.bank_branch_id
       AND srcle.legal_entity_id = cpt.source_legal_entity_id
       AND destba.bank_account_id = cpt.destination_bank_account_id
       AND destbb.branch_party_id = destba.bank_branch_id
       AND destle.legal_entity_id = cpt.destination_legal_entity_id
       AND srcle.party_id = cpt.source_party_id
       AND destle.party_id = cpt.destination_party_id
       AND cpt.trxn_reference_number = :P_BANK_TRANSFER_NUMBER
       AND xte.entity_code = 'CE_CASHFLOWS'
       AND xte.source_id_int_1 = ca.cashflow_id
       AND xte.entity_id = xe.entity_id
       AND xah.ae_header_id = xal.ae_header_id
       AND xah.application_id = xal.application_id
       AND xgl.ledger_id = xah.ledger_id
       AND xlk.lookup_code = xal.accounting_class_code
       AND xlk.lookup_type = 'XLA_ACCOUNTING_CLASS'
       AND get.encumbrance_type_id(+) = xal.encumbrance_type_id
       --             AND xal.ae_header_id = :1
       --             AND xal.application_id = :2
       AND xal.currency_conversion_type = gdct.conversion_type(+)
       AND (   NVL (accounted_cr, 0) <> 0
            OR NVL (accounted_dr, 0) <> 0
            OR fnd_profile.VALUE ('XLA_SHOW_ZERO_AMT_JRNL') = 'Y')
       AND xah.application_id = xe.application_id
       AND xah.event_id = xe.event_id
