SELECT PS_INV.TRX_NUMBER Apply_To, APP.APPLY_DATE APPLY_DATE, APP.AMOUNT_APPLIED AMOUNT_APPLIED
, DECODE(PS_INV.CLASS,'PMT',ART.NAME, DECODE(SIGN(APP.APPLIED_PAYMENT_SCHEDULE_ID), -1,ART.NAME, NULL)) REC_ACTIVITY_NAME
FROM AR_PAYMENT_SCHEDULES_ALL PS_INV,HZ_CUST_ACCOUNTS,HZ_PARTIES,AR_RECEIVABLE_APPLICATIONS_ALL APP,AR_CASH_RECEIPTS_ALL CR
,AR_RECEIVABLES_TRX_ALL ART,RA_CUSTOMER_TRX_LINES_ALL CTL,AR_LOOKUPS L_CLASS,RA_CUSTOMER_TRX_ALL CT,
RA_BATCH_SOURCES_ALL BS
WHERE HZ_CUST_ACCOUNTS.CUST_ACCOUNT_ID           = PS_INV.CUSTOMER_ID
AND   HZ_CUST_ACCOUNTS.PARTY_ID                       = HZ_PARTIES.PARTY_ID
AND  APP.APPLIED_PAYMENT_SCHEDULE_ID = PS_INV.PAYMENT_SCHEDULE_ID
AND APP.DISPLAY                   = 'Y'
AND APP.CASH_RECEIPT_ID             = CR.CASH_RECEIPT_ID
AND ART.RECEIVABLES_TRX_ID(+)       = APP.RECEIVABLES_TRX_ID
AND CTL.CUSTOMER_TRX_LINE_ID(+)     = APP.APPLIED_CUSTOMER_TRX_LINE_ID
AND PS_INV.CLASS                    = L_CLASS.LOOKUP_CODE
AND L_CLASS.LOOKUP_TYPE             = 'INV/CM'
AND CT.CUSTOMER_TRX_ID(+)           = PS_INV.CUSTOMER_TRX_ID
AND BS.BATCH_SOURCE_ID (+)          = CT.BATCH_SOURCE_ID

SELECT  
    fnd_flex_ext.get_segs('SQLGL', 'GL#', gl.chart_of_accounts_id, l.code_combination_id) ACCOUNT ,
    xla_oa_functions_pkg.get_ccid_description (gl.chart_of_accounts_id, l.code_combination_id) account_description ,
    l.code_combination_id ,
    l.entered_dr ,
    l.entered_cr ,
     l.accounted_dr ,
    l.accounted_cr ,
    l.CURRENCY_CODE
    from xla_ae_lines l, xla_gl_ledgers_v gl ,xla_ae_headers h
     WHERE gl.ledger_id                            = h.ledger_id
      AND h.ae_header_id                            = l.ae_header_id
   AND h.application_id                          = l.application_id
   AND (NVL(NVL(accounted_cr,accounted_dr),0)   <> 0
  OR FND_PROFILE.value('XLA_SHOW_ZERO_AMT_JRNL')='Y')
  AND   l.code_combination_id='11055'
