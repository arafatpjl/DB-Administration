SELECT DISTINCT ACR.CASH_RECEIPT_ID,
ACR.TYPE,
ACR.RECEIPT_NUMBER,
ACR.RECEIPT_DATE,
ACR.AMOUNT,
ACR.CURRENCY_CODE,
CE_BANK_AND_ACCOUNT_UTIL.GET_MASKED_BANK_ACCT_NUM(CE_BANK_ACCT_USES_ALL.BANK_ACCOUNT_ID) ACCOUNT_NO_EN,
ACR.RECEIPT_NUMBER AS BANK_REFERENCE_NUMBER,
CBA.BANK_ACCOUNT_NUM_ELECTRONIC ACCOUNT_NUM,
CBB.BANK_NAME,
CBB.BANK_BRANCH_NAME,
HP.PARTY_NAME,
HCA.ACCOUNT_NUMBER PARTY_NUMBER
,ACRH.STATUS
FROM AR_CASH_RECEIPTS_ALL ACR,
AR_CASH_RECEIPT_HISTORY_ALL ACRH,
CE_BANK_ACCT_USES_ALL,
CE_BANK_ACCOUNTS CBA,
CE_BANK_BRANCHES_V CBB,
HZ_CUST_ACCOUNTS HCA,
HZ_PARTIES HP

where CE_BANK_ACCT_USES_ALL.ORG_ID=ACR.ORG_ID 
AND CE_BANK_ACCT_USES_ALL.BANK_ACCT_USE_ID     = ACR.REMIT_BANK_ACCT_USE_ID
AND CE_BANK_ACCT_USES_ALL.BANK_ACCOUNT_ID = CBA.BANK_ACCOUNT_ID
AND CBB.BRANCH_PARTY_ID=CBA.BANK_BRANCH_ID
AND ACR.PAY_FROM_CUSTOMER              = HCA.CUST_ACCOUNT_ID
AND  HCA.PARTY_ID                       = HP.PARTY_ID
AND CE_BANK_ACCT_USES_ALL.BANK_ACCOUNT_ID= CBA.BANK_ACCOUNT_ID
AND ACR.CASH_RECEIPT_ID=ACRH.CASH_RECEIPT_ID
AND ACR.STATUS='APP'
AND ACR.TYPE='CASH'
AND ACR.RECEIPT_NUMBER='BACCHG831095F' 
AND CBA.BANK_ACCOUNT_NUM_ELECTRONIC='004 006623 016'

SELECT CR.CASH_RECEIPT_ID CASH_RECEIPT_ID_INV, 
CR.TYPE,
CR.RECEIPT_NUMBER,
DECODE(ARA_V.trx_number,'Receipt Write-off',rec_activity_name,ARA_V.trx_number) Apply_To,
APP.APPLY_DATE APPLY_DATE,
ARA_V.AMOUNT_APPLIED
FROM AR_PAYMENT_SCHEDULES_ALL PS_INV,
HZ_CUST_ACCOUNTS HCA,
HZ_PARTIES HP,
AR_RECEIVABLE_APPLICATIONS_ALL APP,
AR_CASH_RECEIPTS_ALL CR,
AR_RECEIVABLES_TRX_ALL ART,
RA_CUSTOMER_TRX_LINES_ALL CTL,
AR_LOOKUPS L_CLASS,
RA_CUSTOMER_TRX_ALL CT,
RA_BATCH_SOURCES_ALL BS,
ar_receivable_applications_V ARA_V
WHERE 
CR.CASH_RECEIPT_ID='2000'
AND HCA.CUST_ACCOUNT_ID           = PS_INV.CUSTOMER_ID
AND   HCA.PARTY_ID                       = HP.PARTY_ID
AND  APP.APPLIED_PAYMENT_SCHEDULE_ID = PS_INV.PAYMENT_SCHEDULE_ID
AND APP.DISPLAY                     = 'Y'
AND APP.CASH_RECEIPT_ID             = CR.CASH_RECEIPT_ID
AND ART.RECEIVABLES_TRX_ID(+)       = APP.RECEIVABLES_TRX_ID
AND CTL.CUSTOMER_TRX_LINE_ID(+)     = APP.APPLIED_CUSTOMER_TRX_LINE_ID
AND PS_INV.CLASS                    = L_CLASS.LOOKUP_CODE
AND L_CLASS.LOOKUP_TYPE             = 'INV/CM'
AND CT.CUSTOMER_TRX_ID(+)           = PS_INV.CUSTOMER_TRX_ID
AND BS.BATCH_SOURCE_ID (+)          = CT.BATCH_SOURCE_ID
AND CR.CASH_RECEIPT_ID              = ARA_V.CASH_RECEIPT_ID
AND CR.STATUS='APP'

select 
ACR.CASH_RECEIPT_ID CASH_RECEIPT_ID_ACC,
ACR.TYPE TYPE_ACC,
cba.BANK_ACCOUNT_NUM_ELECTRONIC ACCOUNT_NUMBER_ACC,
xah.accounting_date, 
fnd_flex_ext.get_segs('SQLGL', 'GL#', gl.chart_of_accounts_id, xal.code_combination_id) ACCOUNT ,
xla_oa_functions_pkg.get_ccid_description (gl.chart_of_accounts_id, xal.code_combination_id) account_description ,
xal.ACCOUNTING_CLASS_CODE,
xal.ENTERED_DR,
xal.ENTERED_CR,
xal.accounted_dr ,
xal.accounted_cr
from 
xla_ae_lines xal ,
xla_ae_headers xah,
xla_gl_ledgers_v gl,
xla_events xe,
AR_CASH_RECEIPT_HISTORY_ALL acrh,
AR_CASH_RECEIPTS_ALL ACR,
CE_BANK_ACCT_USES_ALL cbau,
CE_BANK_ACCOUNTS cba
where
xal.application_id = xah.application_id
AND xal.ae_header_id(+) = xah.ae_header_id
AND gl.ledger_id(+)  = xah.ledger_id
AND xah.application_id = xe.application_id
and xah.event_id=xe.event_id
and acrh.event_id=xe.event_id
and ACR.cash_receipt_id=acrh.cash_receipt_id
AND cbau.BANK_ACCT_USE_ID     = ACR.REMIT_BANK_ACCT_USE_ID
AND cbau.BANK_ACCOUNT_ID = cba.BANK_ACCOUNT_ID
AND ACR.STATUS='APP'
and ACR.receipt_number='BACCHG831095F'
and ACR.TYPE='CASH'
and cba.BANK_ACCOUNT_NUM_ELECTRONIC='004 006623 016'
AND CR.STATUS='APP'
