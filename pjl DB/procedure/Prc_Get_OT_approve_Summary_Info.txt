ALTER Procedure [dbo].[Prc_Get_OT_approve_Summary_Info] @ComID Int,@Date SmallDateTime,@Department varchar(30),@Section varchar(30)

--	exec Prc_Get_OT_approve_Summary_Info 1,'14-mar-18','ICT','Software'

AS


select Department,
Section,
DesigAddi,
EmployeeCode,
Name,
DailyDate,
cast('' as varchar(50))  as Shift,
cast('1900-01-01 00:00:00' as datetime)  as OTStart,
cast(0 as int)  as WeekNo,
cast('' as varchar(50))  as TotalWeekOT,
cast(0 as bit)  as WeekLimit,
cast(0 as int)   MonthNo,
cast('' as varchar(50))  as TotalMonthOT  ,
cast(0 as bit)  as MonthLimit,
Wstatus
,
ISNULL(Emp_Daily_Attendance.ValidOT, '00:00') + ISNULL(Emp_Daily_Attendance.RestOT, '00:00') + ISNULL(Emp_Daily_Attendance.ExtraOT, '00:00') AS OT
, ISNULL(B.OTRequ, '00:00') AS OTReque
Into #Emp_OT_Info


from Emp_Daily_Attendance 
 Inner Join  Employee_Information  ON Emp_Daily_Attendance.EmpCode = Employee_Information.EmployeeCode
 FULL OUTER JOIN  DB_Web_Portal.dbo.tblEmpOTRequ B ON Emp_Daily_Attendance.ComID=B.ComID
 AND Emp_Daily_Attendance.EmpCode = B.EmpCode AND Emp_Daily_Attendance.DailyDate = B.Date
 where Department=@Department and Section=@Section  and DailyDate=@Date
 
 SELECT * FROM   #Emp_OT_Info
