----- Wages
 update #tmpPayrollIntegration 
 set amount=Salary.Total
 from #tmpPayrollIntegration
 inner join ( SELECT DeptName, SubDeptName,Total= sum(WEarned+AbsAmount+FoodPay+DinnerAllow+AttenBonus+TransPay+TransAllow+SpAllow+OthersE)  FROM #Salary_And_Wages_Combined 
 group by DeptName, SubDeptName ) as Salary
 on  Salary.DeptName=#tmpPayrollIntegration.DeptName and  Salary.SubDeptName=#tmpPayrollIntegration.SubDeptName
 where  Category='Wages' and  SubCategory='Wages'

select  
		@Comid as ComId
		,Natural_Acc_Code
		,Sub_Acc_Code
		,Category
		,SubCategory	 
		,ERP_COST_CENTER
		,ERP_ShortCode     
		,ERP_CostCenterID
		,sum(Amount) as TotalAmount
		,Type
		,@Month as Month 
		,@Year as Year
		,Getdate() as CreatedDate
	into #tblPayrollIntegration
from #tmpPayrollIntegration  where Amount>0
		group by Natural_Acc_Code
		,Sub_Acc_Code
		,Category
		,SubCategory	 
		,ERP_COST_CENTER
		,ERP_ShortCode     
		,ERP_CostCenterID	
		,Type 


	--------------------
	--------------------


		SELECT
		 ComId
		,Month 
		,Year
		,Natural_Acc_Code
		,Sub_Acc_Code
		,Category
		,SubCategory	 
		,ERP_COST_CENTER
		,ERP_ShortCode     
		,ERP_CostCenterID		
		,isnull(Dr,0) as Dr
		,isnull(Cr,0) as Cr
		,'False' as Status
		,CreatedDate
FROM 
( SELECT B.ERP_CompCode ComId
		,Natural_Acc_Code
		,Sub_Acc_Code
		,Category
		,SubCategory	 
		,ERP_COST_CENTER
		,ERP_ShortCode     
		,ERP_CostCenterID
		,TotalAmount		
		,Type
		,Month 
		,Year
		,CreatedDate
FROM #tblPayrollIntegration 
     A INNER JOIN Company_Information_New B ON A.ComID=B.ID
     ) ps
PIVOT
(
SUM (TotalAmount)
FOR Type IN
( [Dr], [Cr])
) AS pvt

order by Category
		,SubCategory
