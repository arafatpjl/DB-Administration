ALTER Proc [dbo].[prcAlter_viewLeaveRegisterall] @ComID tinyint, @Year smallint, @Month Varchar(15), @EmpStatus Varchar(15)

Begin
	Exec('
	Alter view dbo.viewLeaveRegisterall
	as 
		

SELECT     dbo.Employee_Information.ComID, dbo.Emp_Leave_Entry.EmpCode, dbo.Emp_Leave_Entry.LType, dbo.Emp_Leave_Entry.LDays,

                      dbo.Employee_Information.Name AS empname, dbo.Employee_Information.JoiningDate, dbo.Employee_Information.Department, 
                      dbo.Employee_Information.[Section], dbo.Employee_Information.Designation, UPPER(DATENAME(Month, dbo.Emp_Leave_Entry.FrDate)) AS [Month], 
                      DATEPART(YYYY, dbo.Emp_Leave_Entry.FrDate) AS year
FROM         dbo.Emp_Leave_Entry INNER JOIN
                      dbo.Employee_Information ON dbo.Emp_Leave_Entry.RegID = dbo.Employee_Information.RegID INNER JOIN
                      dbo.Company_Information ON dbo.Employee_Information.ComID = dbo.Company_Information.ID 
                      
WHERE     (dbo.Emp_Leave_Entry.ComID = '+ @ComID +') AND (DATEPART(YYYY, dbo.Emp_Leave_Entry.FrDate) = '+ @Year +') AND (UPPER(DATENAME(Month, 
                      dbo.Emp_Leave_Entry.FrDate)) =  '''+ @Month +''') AND (dbo.Employee_Information.WStatus =''W'' AND dbo.Emp_Leave_Entry.LType NOT IN(''NP'',''NW'') )
