ALTER PROCEDURE [dbo].[PrcExcessOTReport]
AS
Declare @MinDate SMALLDATETIME, @MaxDate SMALLDATETIME, @Month NVARCHAR(20), @Year INT, @MonthID int
SET @Month='APRIL'
SET @Year=2015
SElect @MonthID=SLNo from Month_Name where MONTH=@Month

Select @MinDate= MIN(prDate),@MaxDate= MAX(prDate) From Daily_Process_Status WHERE prMonth=@MonthID AND prYear=@Year


SELECT C.ComID,@Month As Month,@Year AS Year,C.DEPARTMENT,C.Section,COUNT(DISTINCT C.EmpCode) AS EMP,
       SUM(C.EMP40)EMP40,SUM(C.EMP45)EMP45, SUM(C.EMP50)EMP50,SUM(C.EMP50UP)EMP50UP
FROM (SELECT B.ComID,B.Department, B.Section,B.EmpCode,CASE WHEN SUM(B.OT) <=40 THEN
       COUNT(DISTINCT B.EmpCode) ELSE 0 END AS EMP40,CASE WHEN SUM(B.OT) >40 AND SUM(B.OT) <=45 THEN
       COUNT(DISTINCT B.EmpCode) ELSE 0 END AS EMP45,CASE WHEN SUM(B.OT) >45 AND SUM(B.OT) <=50 THEN
       COUNT(DISTINCT B.EmpCode) ELSE 0 END AS EMP50,CASE WHEN SUM(B.OT) >50 THEN
       COUNT(DISTINCT B.EmpCode) ELSE 0 END AS EMP50UP
FROM (SELECT B.ComID,DEPARTMENT,Section,dailydate,EMPCODE,
      (DATEPART(HH,VALIDOT)+DATEPART(HH,RESTOT)+DATEPART(HH,ExtraOT)+
       ((DATEPART(HH,VALIDOT)+DATEPART(HH,RESTOT)+DATEPART(HH,ExtraOT))/60.00)) AS OT
FROM Emp_Daily_Attendance A INNER JOIN Employee_Information B ON 
                   A.ComID = B.ComID AND A.EmpCode = B.EmployeeCode
WHERE (DailyDate BETWEEN @MinDate AND @MaxDate AND WStatus='W')      
GROUP BY B.ComID,DEPARTMENT,Section,dailydate,EMPCODE,ValidOT,RestOT,ExtraOT) B
GROUP BY B.ComID,B.Department, B.Section,B.EmpCode) C
--WHERE Department IN ('SEWING','FINISHING','CUTTING','WET PROCESSING',
--          'PAD','QAD','ENGINEERING','PRINTING','SAMPLE','ADMINISTRATION','MCD','DESIGN & DEVELOPMENT',
--           'STOCK LOT','ADMINISTRATION','ENGINEERING (SEWING)', 'ENGINEERING-SEWING','ENGINEERING-VBM')
GROUP BY C.ComID,C.DEPARTMENT,C.Section
ORDER BY 1,4,5
