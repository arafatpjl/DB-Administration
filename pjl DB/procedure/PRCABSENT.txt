ALTER PROC [dbo].[PRCABSENT] @ComID int, @FromDate Datetime, @ToDate Datetime, @Dept varchar(15), @SubDept Varchar(15)
AS 
--	Declare @ComID int, @FromDate Datetime, @ToDate Datetime, @Dept varchar(15), @SubDept Varchar(15)
--	Select @ComID = 1, @FromDate = '04-11-05', @ToDate = '04-11-05', @Dept = 'FINISHING', @SubDept =  'FINISHING (01)'

--	TRUNCATE TABLE TEMP_EMP_DAILY_ATTEN

	declare @Count int, @Date datetime

	Declare Cursor_Date scroll Cursor
	for Select DISTINCT prDate From Daily_Process_Status where prDate Between @FromDate and @ToDate
	open Cursor_Date

	select @Count = @@Cursor_rows

	While @Count> 0
	begin
		Fetch next from Cursor_Date into @Date

		INSERT TEMP_EMP_DAILY_ATTEN (COMID, DAILYDATE, EMPCODE, EMPNAME, EMPDESIG, DEPTNAME, SUBDEPT, TIMEIN, TIMEOUT, OTHRS, RESTOT, EMPSTATUS )
		SELECT  'COMID' = @COMID, 'DAILYDATE' = @Date, EMPLOYEECODE, NAME, DESIGNATION, 'DEPT' = @DEPT, 'SUBDEPT' = @SUBDEPT, 'TIMEIN'="00:00", 'TIMEOUT'="00:00",  'VALIDOT'="00:00", 'RESTOT'="00:00", 'EMPSTATUS'= 'A'
		FROM EMPLOYEE_INFORMATION
		WHERE MLEAVE = 0 AND MLEFT = 0 AND DEPARTMENT = @DEPT AND SECTION = @SUBDEPT

		UPDATE TEMP_EMP_DAILY_ATTEN
		SET TIMEIN = (SELECT MIN(T.DAILYTIME) AS 'TIMEIN' FROM Temp_Emp_Daily_Atten A INNER JOIN EMP_DAILY_TIME T ON A.COMID = T.COMID AND A.EMPCODE = T.EMPCODE AND A.DAILYDATE = T.DAILYDATE WHERE T.DAILYDATE =  @DATE )
		FROM Temp_Emp_Daily_Atten A 
		INNER JOIN EMP_DAILY_TIME T ON A.COMID = T.COMID AND A.EMPCODE = T.EMPCODE  AND A.DAILYDATE = T.DAILYDATE
		WHERE T.DAILYDATE =  @DATE   
	
		UPDATE TEMP_EMP_DAILY_ATTEN
		SET TIMEOUT = (SELECT MAX(T.DAILYTIME) AS 'TIMEOUT' FROM Temp_Emp_Daily_Atten A INNER JOIN EMP_DAILY_TIME T ON A.COMID = T.COMID AND A.EMPCODE = T.EMPCODE AND A.DAILYDATE = T.DAILYDATE WHERE T.DAILYDATE = @DATE )
		FROM Temp_Emp_Daily_Atten A 
		INNER JOIN EMP_DAILY_TIME T ON A.COMID = T.COMID AND A.EMPCODE = T.EMPCODE AND A.DAILYDATE = T.DAILYDATE
		WHERE T.DAILYDATE = @DATE 

		UPDATE TEMP_EMP_DAILY_ATTEN
		SET EMPSTATUS = 'P' 
		FROM Temp_Emp_Daily_Atten A 
		INNER JOIN EMP_DAILY_TIME T ON A.COMID = T.COMID AND A.EMPCODE = T.EMPCODE AND A.DAILYDATE = T.DAILYDATE
		WHERE T.DAILYDATE = @DATE  

		SELECT @COUNT = @COUNT - 1
END
CLOSE Cursor_Date
DEALLOCATE Cursor_Date

DELETE TEMP_EMP_DAILY_ATTEN
WHERE EMPSTATUS = 'P'
