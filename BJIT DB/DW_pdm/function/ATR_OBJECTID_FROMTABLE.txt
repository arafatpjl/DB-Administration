create or replace FUNCTION ATR_OBJECTID_FROMTABLE (tablename IN VARCHAR2, lxoid IN NUMBER) RETURN VARCHAR2 IS
  v_vaultid NUMBER;
  v_lxoid NUMBER;
  v_tempid NUMBER;
  v_next_digit NUMBER;
  v_result VARCHAR(30);
  v_tempchar VARCHAR2(50);
  objectid1 VARCHAR(6);
  objectid2 VARCHAR(6);
  objectid3 VARCHAR(6);
  objectid4 VARCHAR(6);
BEGIN
  v_tempid := INSTRB(tablename, '_', 1, 1) + 1;
  v_tempchar := SUBSTR(tablename, v_tempid);
  v_vaultid := Atr_Hex2dec(v_tempchar);
  v_lxoid := lxoid;

  IF v_vaultid <0 THEN
     v_vaultid := 4294967296 + v_vaultid;
  END IF;
  v_vaultid := v_vaultid / 65536;
  v_tempid := FLOOR(v_vaultid);
  objectid1 := TO_CHAR(v_tempid);
  v_vaultid :=v_vaultid - v_tempid;
  v_vaultid := v_vaultid * 65536;
  objectid2 := TO_CHAR(FLOOR(v_vaultid));

  IF v_lxoid <0 THEN
     v_lxoid := 4294967296 + v_lxoid;
  END IF;
  v_lxoid := v_lxoid / 65536;
  v_tempid := FLOOR(v_lxoid);
  objectid3 := TO_CHAR(v_tempid);
  v_lxoid :=v_lxoid - v_tempid;
  v_lxoid := v_lxoid * 65536;
  objectid4 := TO_CHAR(FLOOR(v_lxoid));

  v_result := objectid1||'.'||objectid2||'.'||objectid3||'.'||objectid4;
  RETURN v_result;
END Atr_Objectid_Fromtable;
