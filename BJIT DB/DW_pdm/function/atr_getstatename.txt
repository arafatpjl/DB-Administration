create or replace FUNCTION atr_getstatename (in_username VARCHAR2, in_mxpolicyoid NUMBER, in_mxstateoid NUMBER) RETURN VARCHAR2 AS
    sAttrValue VARCHAR2(255);

BEGIN


     EXECUTE IMMEDIATE 'SELECT MXNAME FROM '||in_username||'.MXSTATEREQ WHERE mxoid=:1 AND MXPOLICY =:2'  INTO sAttrValue using in_mxstateoid, in_mxpolicyoid;

     RETURN sAttrValue;
END;
