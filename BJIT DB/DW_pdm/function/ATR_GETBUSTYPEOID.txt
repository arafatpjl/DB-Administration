create or replace FUNCTION ATR_GETBUSTYPEOID (in_username VARCHAR2, sAttrName VARCHAR2) RETURN NUMBER AS
    sAttrValue VARCHAR2(255);

BEGIN


     EXECUTE IMMEDIATE 'SELECT mxoid FROM '||in_username||'.MXBUSTYPE WHERE MXNAME=:1'  INTO sAttrValue using sAttrName;

     RETURN sAttrValue;
END;
