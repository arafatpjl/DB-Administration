create or replace FUNCTION atr_getpolicyname (in_username VARCHAR2, in_mxoid NUMBER) RETURN VARCHAR2 AS
    sAttrValue VARCHAR2(255);

BEGIN


     EXECUTE IMMEDIATE 'SELECT MXNAME FROM '||in_username||'.MXPOLICY WHERE mxoid=:1'  INTO sAttrValue using in_mxoid;

     RETURN sAttrValue;
END;
