create or replace PROCEDURE dw_bom_u (inUsername IN VARCHAR2, InIntegID IN INTEGER, InVaultId IN NUMBER, InLogId IN NUMBER) AS

 UserName VARCHAR2(20);
 VaultName VARCHAR2(128);

 TableName VARCHAR2(60);
 TableName2 VARCHAR2(60);

 TYPE CurTyp IS REF CURSOR;

 c_bus CurTyp;
 cursor_str VARCHAR2(200);


 logmessage VARCHAR2(255);


 v_lxoid number;
 v_lxvault number;

 iTotalNumber number;
 ilooptest number;

 s_val1 varchar2(255);
 s_val2 varchar2(255);
 s_val3 varchar2(255);
 s_attribute_name varchar2(255);
 s_sql_string varchar2(1000);
 log_ID NUMBER := 0;
 icheck NUMBER;
 sLattice VARCHAR2(20);
 v_CommonTextId NUMBER;
 v_PurchasingTextId NUMBER;
 v_ManufacturingTextId NUMBER;



BEGIN


	UserName := UPPER(inUsername);
	sLattice :=  Atr_Dec_To_Hex(InVaultId);

        -- lxoids ofattributes

	v_CommonTextId := ATR_GETATTROID(inUsername,'Common text');
        v_PurchasingTextId := ATR_GETATTROID(inUsername,'Purchasing text');
        v_ManufacturingTextId := ATR_GETATTROID(inUsername,'Manufacturing text');

	TableName2 := UserName||'.LXRO_'||sLattice;
        EXECUTE IMMEDIATE 'TRUNCATE TABLE ATR_LXOID_TEMP1';
        COMMIT;

              -- Selecting and Inserting relations

              s_sql_string := 'INSERT INTO INTEG_RELS (INTEG_ID, PARENT_O_ID, CHILD_O_ID, REL_ID, REL_NAME) ';
              s_sql_string := s_sql_string||'SELECT :1, ATR_OBJECTID(:2,T1.LXFROMID), ATR_OBJECTID(T1.LXTOLAT,T1.LXTOID), ATR_OBJECTID(:3,T1.LXOID), T2.REL_TYPE';
              s_sql_string := s_sql_string||' FROM  '||TableName2||' T1, INTEG_DW_RELS T2';
              s_sql_string := s_sql_string||' WHERE T1.LXFROMLAT = :4 ';
              s_sql_string := s_sql_string||' AND T1.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :5) ';
              s_sql_string := s_sql_string||' AND T1.LXFLAGS <> -1';
              s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
	      s_sql_string := s_sql_string||' AND T1.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)';


              EXECUTE IMMEDIATE s_sql_string using InIntegID, InVaultId, InVaultId, InVaultId, InVaultId;
              COMMIT;

	      -- Selecting and Inserting real attributes of relations

   	      TableName := UserName||'.LXREAL_'||sLattice;
              s_sql_string := 'INSERT INTO INTEG_REL_ATTR_REAL (INTEG_ID, REL_ID, ATTR_NAME, ATTR_ID, ATTR_VALUE)';
              s_sql_string := s_sql_string||' SELECT :1, ATR_OBJECTID(:2,T1.LXOID), T2.ATTR_NAME , T1.LXTYPE, T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	      s_sql_string := s_sql_string||' WHERE T1.LXOID IN (SELECT LXOID FROM '||TableName2||' T3, INTEG_DW_RELS T4 WHERE T3.LXFROMLAT = :3 AND T3.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :4) AND T3.LXTYPE = T4.LXTYPE AND T3.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)  )';
              s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
              s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 1)';
              EXECUTE IMMEDIATE s_sql_string using InIntegID, InVaultId,InVaultId, InVaultId;

	      COMMIT;

              -- Selecting and Inserting integer attributes of relations

   	      TableName := UserName||'.LXINT_'||sLattice;
              s_sql_string := 'INSERT INTO INTEG_REL_ATTR_INTEGER (INTEG_ID,REL_ID, ATTR_NAME, ATTR_ID, ATTR_VALUE)';
              s_sql_string := s_sql_string||' SELECT :1, ATR_OBJECTID(:2,T1.LXOID), T2.ATTR_NAME , T1.LXTYPE, T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	      s_sql_string := s_sql_string||' WHERE T1.LXOID IN (SELECT LXOID FROM '||TableName2||' T3, INTEG_DW_RELS T4 WHERE T3.LXFROMLAT = :3 AND T3.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :4) AND T3.LXTYPE = T4.LXTYPE AND T3.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)  )';
              s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
              s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 0)';
              EXECUTE IMMEDIATE s_sql_string using InIntegID, InVaultId,InVaultId, InVaultId;


	      COMMIT;

              -- Selecting and Inserting date attributes of relations

   	      TableName := UserName||'.LXDATE_'||sLattice;
              s_sql_string := 'INSERT INTO INTEG_REL_ATTR_DATE (INTEG_ID, REL_ID, ATTR_NAME, ATTR_ID, ATTR_VALUE)';
              s_sql_string := s_sql_string||' SELECT :1, ATR_OBJECTID(:2,T1.LXOID), T2.ATTR_NAME , T1.LXTYPE, T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	      s_sql_string := s_sql_string||' WHERE T1.LXOID IN (SELECT LXOID FROM '||TableName2||' T3, INTEG_DW_RELS T4 WHERE T3.LXFROMLAT = :3 AND T3.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :4) AND T3.LXTYPE = T4.LXTYPE AND T3.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)  )';
              s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
              s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 3)';
              EXECUTE IMMEDIATE s_sql_string using InIntegID, InVaultId,InVaultId,InVaultId;

	      COMMIT;

	      -- Selecting and Inserting boolean attributes of relations

   	      TableName := UserName||'.LXBOOL_'||sLattice;
              s_sql_string := 'INSERT INTO INTEG_REL_ATTR_BOOLEAN (INTEG_ID,REL_ID, ATTR_NAME, ATTR_ID, ATTR_VALUE)';
              s_sql_string := s_sql_string||' SELECT :1, ATR_OBJECTID(:2,T1.LXOID), T2.ATTR_NAME , T1.LXTYPE, T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	      s_sql_string := s_sql_string||' WHERE T1.LXOID IN (SELECT LXOID FROM '||TableName2||' T3, INTEG_DW_RELS T4 WHERE T3.LXFROMLAT = :3 AND T3.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :4) AND T3.LXTYPE = T4.LXTYPE AND T3.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)  )';
              s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
              s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 4)';
              EXECUTE IMMEDIATE s_sql_string using InIntegID, InVaultId, InVaultId, InVaultId;

	      COMMIT;

	      -- Selecting and Inserting string attributes of relations

   	      TableName := UserName||'.LXSTRING_'||sLattice;
              s_sql_string := 'INSERT INTO INTEG_REL_ATTR_STRING (INTEG_ID, REL_ID, ATTR_NAME, ATTR_ID, ATTR_VALUE)';
              s_sql_string := s_sql_string||' SELECT :1, ATR_OBJECTID(:2,T1.LXOID), T2.ATTR_NAME , T1.LXTYPE, T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	      s_sql_string := s_sql_string||' WHERE T1.LXOID IN (SELECT LXOID FROM '||TableName2||' T3, INTEG_DW_RELS T4 WHERE T3.LXFROMLAT = :3 AND T3.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :4) AND T3.LXTYPE = T4.LXTYPE AND T3.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)  )';

	      s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
              s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 2)';
	      s_sql_string := s_sql_string||' AND T1.LXVAL IS NOT NULL';
              EXECUTE IMMEDIATE s_sql_string using InIntegID, InVaultId,InVaultId, InVaultId;

	      COMMIT;

	      -- Selecting and Inserting BOM texts of relations

              s_sql_string := 'INSERT INTO T_DW_TEXTS (LXOID,LXTYPE,LXVAULT,LXVAL1,LXVAL3,O_ID, DEPENDENCY_TYPE,O_ATTR_NAME,O_ATTR_NAME_PREFIX)';
              s_sql_string := s_sql_string||' SELECT T1.LXOID, T1.LXTYPE, :1, T1.LXVAL, T1.LXVAL, ATR_OBJECTID(:2,T1.LXOID), :3, T2.MXNAME, :4 ';
              s_sql_string := s_sql_string||' FROM '||Tablename||' T1, '||UserName||'.MXATTRTYPE T2  ';
              s_sql_string := s_sql_string||' WHERE T1.LXOID IN (SELECT LXOID FROM '||TableName2||' T3, INTEG_DW_RELS T4 WHERE T3.LXFROMLAT = :5 AND T3.LXFROMID IN (SELECT LXOID FROM INTEG_OBJ WHERE LXVAULT = :6) AND T3.LXTYPE = T4.LXTYPE AND T3.LXTYPE IN (SELECT LXTYPE FROM INTEG_DW_RELS)  )';

              s_sql_string := s_sql_string||' AND  LXTYPE IN ('||v_CommonTextId||','||v_PurchasingTextId||','||v_PurchasingTextId||') ';
              s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.MXOID ';
              s_sql_string := s_sql_string||' AND LXVAL IS NOT NULL';

	      EXECUTE IMMEDIATE s_sql_string using InVaultId, InVaultId, 'BOM texts', 'BOM ', InVaultId, InVaultId;
              COMMIT;


	COMMIT;

END;
