create or replace PROCEDURE dw_bom_substitute (inUsername IN VARCHAR2, InIntegID IN INTEGER, InVaultId IN NUMBER, InVaultName IN VARCHAR2,InLogId IN NUMBER) AS

 UserName VARCHAR2(20);
 VaultName VARCHAR2(128);
 TableName VARCHAR2(60);
 TYPE CurTyp IS REF CURSOR;
 c1 CurTyp;
 c2 CurTyp;
 cursor_str1 VARCHAR2(200);
 cursor_str2 VARCHAR2(200);
 v_Oid1 VARCHAR2(40);
 v_Oid2 VARCHAR2(40);
 logmessage VARCHAR2(255);
 iTotalNumber number;
 ilooptest number;
 s_sql_string varchar2(1000);
 log_ID NUMBER := 0;
 sLattice VARCHAR2(20);

 v_RelTypelxoid NUMBER;
 v_str1 varchar2(2000);

 v_fromlxoid number;
 v_fromVaultlxoid number;
 v_rellxoid number;
 v_tolxoid number;
 v_toVaultlxoid number;

 v_from_id varchar2(40);
 v_toid varchar2(40);
 v_relid varchar2(40);
 v_rel_attr_name varchar2(40);
 v_rowid ROWID;


 err_msg varchar2(1000);



BEGIN
	log_ID := InLogId;
 	logmessage := 'dw_bom_substitute : Vault : '||InVaultName||' is started ...';
 	Atr_Write_Script_Log('dw_main',logmessage,log_ID);

	UserName := UPPER(inUsername);
	sLattice :=  Atr_Dec_To_Hex(InVaultId);

	v_RelTypelxoid := ATR_GETRELTYPEOID(inUsername,'BOM related spare part');


        EXECUTE IMMEDIATE 'TRUNCATE TABLE ATR_REL_LXOID_TEMP1';

        COMMIT;

 	IF InIntegID = 0 THEN
           BEGIN

  	      logmessage := 'dw_bom_substitute : Vault : '||InVaultName||' , Relations Substitute item -> ATR_REL_LXOID_TEMP1 ...';
 	      Atr_Write_Script_Log('dw_main',logmessage,log_ID);
              TableName := UserName||'.LXRO_'||sLattice;
	      s_sql_string := 'INSERT INTO ATR_REL_LXOID_TEMP1 (PARENT_LXOID, PARENT_VAULT_LXOID, REL_LXOID, REL_ID, PARENT_O_ID, CHILD_O_ID, CHILD_LXOID, CHILD_VAULT_LXOID, LXVAL1) ';
              s_sql_string := s_sql_string||'SELECT T1.LXFROMID, T1.LXFROMLAT, T1.LXOID, ATR_OBJECTID(T1.LXFROMLAT,T1.LXFROMID), ATR_OBJECTID(T1.LXFROMLAT,T1.LXFROMID), ATR_OBJECTID(T1.LXTOLAT,T1.LXTOID), T1.LXTOID, T1.LXTOLAT ,:1';
              s_sql_string := s_sql_string||' FROM  '||TableName||' T1';
              s_sql_string := s_sql_string||' WHERE T1.LXTYPE = :2';
              s_sql_string := s_sql_string||' AND T1.LXFROMLAT = :3';
              EXECUTE IMMEDIATE s_sql_string using 'BOM related spare part', v_RelTypelxoid,InVaultId;
              COMMIT;
 	      logmessage := 'dw_bom_substitute : Vault : '||InVaultName||' , Relations Substitute item -> ATR_REL_LXOID_TEMP1 completed';
 	      Atr_Write_Script_Log('dw_main',logmessage,log_ID);

	      logmessage := 'dw_bom_substitute  - copying attribute Substitute Item : Vault : '||InVaultName||' is started ...';
 	      Atr_Write_Script_Log('dw_main',logmessage,log_ID);

              cursor_str1 := 'SELECT ROWID, PARENT_LXOID, PARENT_VAULT_LXOID, REL_LXOID, REL_ID, PARENT_O_ID, CHILD_O_ID, CHILD_LXOID, CHILD_VAULT_LXOID, LXVAL1 FROM ATR_REL_LXOID_TEMP1';
              OPEN c1 FOR cursor_str1;
	      ilooptest := 1;
              iTotalNumber := 0;
              LOOP
	        FETCH c1 INTO v_rowid, v_fromlxoid,v_fromVaultlxoid,v_rellxoid,v_relid,v_from_id,v_toid,v_tolxoid,v_toVaultlxoid,v_rel_attr_name;

	          EXIT WHEN c1%NOTFOUND;
                    BEGIN

                             s_sql_string := 'INSERT INTO INTEG_RELS_A (PARENT_O_ID,CHILD_O_ID,REL_ID,REL_TYPE_ID,REL_NAME,REL_LXOID,REL_VAULT_LXOID) ';
                             s_sql_string := s_sql_string||' VALUES (:1,:2,:3,:4,:5,:6,:7) ';
                             EXECUTE IMMEDIATE s_sql_string using v_from_id,v_toid,v_relid,v_RelTypelxoid,v_rel_attr_name,v_rellxoid,v_fromVaultlxoid;
                             COMMIT;

                             TableName := UserName||'.LXINT_'||sLattice;
                             s_sql_string := 'INSERT INTO INTEG_REL_ATTR_INTEGER_A (REL_ID, ATTR_NAME, ATTR_VALUE)';
                             s_sql_string := s_sql_string||' SELECT ATR_OBJECTID(:1,T1.LXOID), T2.ATTR_NAME , T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	                     s_sql_string := s_sql_string||' WHERE T1.LXOID = :2';
                             s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
                             s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 0)';
                             EXECUTE IMMEDIATE s_sql_string using InVaultId,v_rellxoid;
			     COMMIT;

                             TableName := UserName||'.LXSTRING_'||sLattice;
                             s_sql_string := 'INSERT INTO INTEG_REL_ATTR_STRING_A (REL_ID, ATTR_NAME, ATTR_VALUE)';
                             s_sql_string := s_sql_string||' SELECT ATR_OBJECTID(:1,T1.LXOID), T2.ATTR_NAME , T1.LXVAL FROM '||TableName||' T1, INTEG_DW_REL_ATTRS T2 ';
	                     s_sql_string := s_sql_string||' WHERE T1.LXOID = :2';
                             s_sql_string := s_sql_string||' AND T1.LXTYPE = T2.LXTYPE';
                             s_sql_string := s_sql_string||' AND T2.LXTYPE IN (SELECT T4.LXTYPE FROM INTEG_DW_REL_ATTRS T4 WHERE T4.ATTR_TYPE_ID = 2)';
	                     s_sql_string := s_sql_string||' AND T1.LXVAL IS NOT NULL';
                             EXECUTE IMMEDIATE s_sql_string using InVaultId,v_rellxoid;
                             COMMIT;

                    END;
                    ilooptest := ilooptest+ 1;
                    iTotalNumber := iTotalNumber + 1;
                    IF ilooptest > 1000 THEN
                       logmessage := 'dw_bom_substitute  - copying attributes  '||to_char(iTotalNumber)||' so far : Vault : '||InVaultName||' ... ';
    	               Atr_Write_Script_Log('dw_main',logmessage,log_ID);
                       COMMIT;
                       ilooptest := 1;
                    END IF;
                   COMMIT;
              END LOOP;

              logmessage := 'dw_bom_substitute  - copying attributes Total '||to_char(iTotalNumber)||' pcs : Vault : '||InVaultName||'  is completed';
    	      Atr_Write_Script_Log('dw_main',logmessage,log_ID);
              COMMIT;

           END;
         ELSE
           BEGIN
	  	NULL;
	   END;
       END IF;

END;
